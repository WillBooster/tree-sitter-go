package tags
type Record struct {
 Name string `json:"name"custom-tag:"value"`
 Broken string `json:"unterminated`
 Arbitrary string `plain : "other text`
 Spaced string `json: "value"`
 Quoted string "json:\"name\""
 Empty string ``
}

type Boundary struct { First string `a:"unfinished`; Second string `b:"second"` }
type EscapedBoundary struct { First string `a:"unfinished\`; Second string `b:"second"` }
